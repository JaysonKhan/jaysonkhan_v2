# jaysonkhan_v2 server-monitor crontab
#
# Install on the server (one-time, as the user that owns /var/www/jaysonkhan):
#
#   crontab -e
#   # paste the lines below, save
#
# Wrapper command `cron_run` records every execution into the
# `ops.CronRun` table (visible at /jk-dinadmin/ops/cronrun/), so the
# admin panel always shows the latest run + status, without you needing
# to dig through journalctl.
#
# All crons share the same prefix block — change it once if paths move.

PROJECT_DIR=/var/www/jaysonkhan
PYTHON=/var/www/jaysonkhan/backend/venv/bin/python
MANAGE=/var/www/jaysonkhan/backend/manage.py
LOG=/var/log/jaysonkhan/cron.log

# ── Server health monitoring ─────────────────────────────────────────────────
# STAGGERING: no job fires at :00 (avoids thundering-herd when UzExam
# hourly jobs also start at :00, causing all-cores-100% CPU burst).
# These values match install-servermonitor-cron.sh exactly.

# Every 10 minutes — CPU core overload alert (≥85% / 5s sustained sample)
3,13,23,33,43,53 * * * *   cd $PROJECT_DIR/backend && DJANGO_SETTINGS_MODULE=$DJ_SETTINGS $PYTHON $MANAGE cron_run check_cpu_alert      >> $LOG 2>&1

# Every 2 minutes (odd minutes — never :00) — per-service state-change
# detection + Telegram alert. Near-real-time; cheap (batched systemctl).
1-59/2 * * * *   cd $PROJECT_DIR/backend && DJANGO_SETTINGS_MODULE=$DJ_SETTINGS $PYTHON $MANAGE cron_run service_health_check >> $LOG 2>&1

# Hourly at :02 — failed and overdue managed crons
2 * * * *     cd $PROJECT_DIR/backend && DJANGO_SETTINGS_MODULE=$DJ_SETTINGS $PYTHON $MANAGE cron_run cron_health_check    >> $LOG 2>&1

# Daily at 09:00 Berlin (11:00 Tashkent) — full server health report
0 9 * * *     cd $PROJECT_DIR/backend && DJANGO_SETTINGS_MODULE=$DJ_SETTINGS $PYTHON $MANAGE cron_run server_health_report >> $LOG 2>&1
